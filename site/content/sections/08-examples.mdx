## Examples

Each example runs against [the schema at the top of the page](#schema), and calls the client `db`. Pick a dialect once, and every example follows it.

### List rows

A query that can match more than one row resolves to an array.

```sql tab="list_users.sql" dialect="all"
select id, name, role, tags, created_at from users
order by created_at desc
limit :limit::int
```

```ts twoslash tab="main.ts"
import listUsers from "./list_users.sql";

const users = await listUsers.run(db, { limit: 20 });
//    ^?
```

### Count rows

An aggregate without `group by` is exactly one row.

```sql tab="count_posts.sql" dialect="all"
select count(*) as total from posts where user_id = :user_id::int
```

```ts twoslash tab="main.ts"
import countPosts from "./count_posts.sql";

const { total } = await countPosts.run(db, { user_id: 1 });
//      ^?
```

### Filter on an optional value

A `coalesce` lets a parameter be `null`: here `null` matches every user. A parameter takes `null` only where the query allows it, such as a `coalesce` or a nullable column it is written to.

```sql tab="posts_by_user.sql" dialect="all"
select id, title from posts
where user_id = coalesce(:user_id::int, user_id)
```

```ts twoslash tab="main.ts"
import postsByUser from "./posts_by_user.sql";

const posts = await postsByUser.run(db, { user_id: null });
//                                        ^?
```

### Sort by a chosen column

A parameter is a value, never a column name, so choose the column with `case`:

```sql tab="sorted_users.sql" dialect="all"
select id, name, created_at from users
order by
  case :sort::text when 'name' then name end,
  created_at desc
```

```ts twoslash tab="main.ts"
import sortedUsers from "./sorted_users.sql";

const users = await sortedUsers.run(db, { sort: "name" });
//    ^?
```

### Search several columns

Each parameter appears once in a query. To use a value twice, select it once in a `with`:

```sql tab="search_users.sql" dialect="all"
with input as (
  select concat('%', :q::text, '%') as pattern
)
select id, name from users, input
where name like input.pattern or email like input.pattern
```

```ts twoslash tab="main.ts"
import searchUsers from "./search_users.sql";

const users = await searchUsers.run(db, { q: "ada" });
//    ^?
```

### Join tables

A column of the table a `left join` adds can be `null`, and a `coalesce` gives it a default.

```sql tab="user_posts.sql" dialect="all"
select u.name, p.title, coalesce(p.title, '') as label
from users u
left join posts p on p.user_id = u.id
```

```ts twoslash tab="main.ts"
import userPosts from "./user_posts.sql";

const rows = await userPosts.run(db);
//    ^?
```

### Page through rows

Pass the sort key of the last row you have, here `created_at` and `id`, to get the rows after it.

```sql file="next_page.sql" dialect="all"
select id, name, created_at from users
where (created_at, id) < (:created_at, :id)
order by created_at desc, id desc
limit 20
```

<DialectTabs follow>
<Dialect value="Postgres">

```ts twoslash
import nextPage from "./next_page.sql";

// the first page: a cursor after every row
const page = await nextPage.run(db, {
  created_at: "3000-01-01 00:00:00+00:00",
  id: 0,
});
const next = await nextPage.run(db, page.at(-1)!);
//    ^?
```

</Dialect>

### Filter by a list

Pass the list as one parameter.

```sql file="users_by_ids.sql" tab="Postgres" tab-group="dialect"
select id, name from users where id = any(:ids::int[])
```

```sql file="users_by_ids.sql" tab="MySQL" tab-group="dialect"
with input as (
  select :ids::json('{
    "type": "array",
    "items": { "type": "integer" }
  }') as ids
)
select id, name from users
where id in (
  select j.id from input, json_table(input.ids, '$[*]' columns (
    id int path '$'
  )) as j
)
```

```sql file="users_by_ids.sql" tab="SQLite" tab-group="dialect"
with input as (
  select :ids::json('{
    "type": "array",
    "items": { "type": "integer" }
  }') as ids
)
select id, name from users
where id in (select j.value from input, json_each(input.ids) as j)
```

```ts twoslash
import usersByIds from "./users_by_ids.sql";

const users = await usersByIds.run(db, { ids: [1, 2, 3] });
//                                       ^?
```

### Keep a list's order

Number the list's elements, and sort by the number.

```sql file="users_in_order.sql" tab="Postgres" tab-group="dialect"
-- with ordinality numbers the elements
select u.id, u.name
from unnest(:ids::int[]) with ordinality as i(id, position)
join users u on u.id = i.id
order by i.position
```

```sql file="users_in_order.sql" tab="MySQL" tab-group="dialect"
-- a for ordinality column numbers the elements
with input as (
  select :ids::json('{
    "type": "array",
    "items": { "type": "integer" }
  }') as ids
)
select u.id, u.name
from input, json_table(input.ids, '$[*]' columns (
  position for ordinality,
  id int path '$'
)) as i
join users u on u.id = i.id
order by i.position
```

```sql file="users_in_order.sql" tab="SQLite" tab-group="dialect"
-- json_each numbers the elements as key
with input as (
  select :ids::json('{
    "type": "array",
    "items": { "type": "integer" }
  }') as ids
)
select u.id, u.name
from input, json_each(input.ids) as i
join users u on u.id = i.value
order by i.key
```

```ts twoslash
import usersInOrder from "./users_in_order.sql";

const users = await usersInOrder.run(db, { ids: [3, 1, 2] });
//    ^?
```

### Nest related rows

Build the related rows as JSON in the statement. Reylo types the JSON from the arguments that build it.

```sql file="user_with_posts.sql" tab="Postgres" tab-group="dialect"
select u.id, u.name,
  coalesce((
    select jsonb_agg(jsonb_build_object(
      'id', p.id,
      'title', p.title,
      'at', p.published_at
    ))
    from posts p where p.user_id = u.id
  ), '[]') as posts
from users u
where u.id = :id::int
```

```sql file="user_with_posts.sql" tab="MySQL" tab-group="dialect"
select u.id, u.name,
  coalesce((
    select json_arrayagg(json_object(
      'id', p.id,
      'title', p.title,
      'at', p.published_at
    ))
    from posts p where p.user_id = u.id
  ), json_array()) as posts
from users u
where u.id = :id::int
```

```sql file="user_with_posts.sql" tab="SQLite" tab-group="dialect"
select u.id, u.name,
  (
    select json_group_array(json_object(
      'id', p.id,
      'title', p.title,
      'at', p.published_at
    ))
    from posts p where p.user_id = u.id
  ) as posts
from users u
where u.id = :id::int
```

```ts twoslash
import userWithPosts from "./user_with_posts.sql";

const user = await userWithPosts.run(db, { id: 1 });
//    ^?
```

<Dialect value="SQLite">

SQLite returns the JSON it builds as text, so `posts` is a string.

</Dialect>

### Insert many rows

Pass the rows as one [JSON parameter](#json-parameters): the schema in its cast types them, and Reylo checks every row before the statement runs.

```sql file="create_users.sql" tab="Postgres" tab-group="dialect"
with input as (
  select :users::jsonb('{
    "type": "array",
    "maxItems": 1000,
    "items": {
      "type": "object",
      "properties": {
        "name": { "type": "string" },
        "email": { "type": ["string", "null"], "format": "email" }
      },
      "required": ["name"]
    }
  }') as users
)
insert into users (name, email)
select u.name, u.email
from input, jsonb_to_recordset(input.users) as u(name text, email text)
returning id
```

```sql file="create_users.sql" tab="MySQL" tab-group="dialect"
insert into users (name, email)
with input as (
  select :users::json('{
    "type": "array",
    "maxItems": 1000,
    "items": {
      "type": "object",
      "properties": {
        "name": { "type": "string" },
        "email": { "type": ["string", "null"], "format": "email" }
      },
      "required": ["name"]
    }
  }') as users
)
select u.name, u.email
from input, json_table(input.users, '$[*]' columns (
  name text path '$.name',
  email varchar(255) path '$.email'
)) as u
```

```sql file="create_users.sql" tab="SQLite" tab-group="dialect"
with input as (
  select :users::json('{
    "type": "array",
    "maxItems": 1000,
    "items": {
      "type": "object",
      "properties": {
        "name": { "type": "string" },
        "email": { "type": ["string", "null"], "format": "email" }
      },
      "required": ["name"]
    }
  }') as users
)
insert into users (name, email)
select u.value ->> 'name', u.value ->> 'email'
from input, json_each(input.users) as u
returning id
```

### Insert related rows

Users with their posts, in one parameter. The posts find their user by `email`, a unique key.

<DialectTabs>
<Dialect value="Postgres">

Postgres writes both tables in one statement: a `with` inserts the users, and the statement inserts the posts.

```sql file="create_users_with_posts.sql"
with input as (
  select :users::jsonb('{
    "type": "array",
    "items": {
      "type": "object",
      "properties": {
        "name": { "type": "string" },
        "email": { "type": "string", "format": "email" },
        "posts": {
          "type": "array",
          "items": {
            "type": "object",
            "properties": { "title": { "type": "string" } },
            "required": ["title"]
          }
        }
      },
      "required": ["name", "email", "posts"]
    }
  }') as users
),
new_users as (
  insert into users (name, email)
  select u.name, u.email
  from input, jsonb_to_recordset(input.users) as u(name text, email text)
  returning id, email
)
insert into posts (user_id, title)
select n.id, p.title
from input
cross join jsonb_to_recordset(input.users) as u(email text, posts jsonb)
join new_users n on n.email = u.email
cross join jsonb_to_recordset(u.posts) as p(title text)
```

```ts twoslash
import createUsersWithPosts from "./create_users_with_posts.sql";
import type { Params } from "./create_users_with_posts.sql";
//            ^?

const users = [
  {
    name: "Ada",
    email: "ada@example.com",
    posts: [{ title: "Notes" }],
  },
];
await createUsersWithPosts.run(db, { users });
```

</Dialect>
<Dialect value="MySQL">

On MySQL, a `with` only reads, so the users and the posts are two statements, run in one transaction:

```sql title="insert_users.sql"
insert into users (name, email)
with input as (
  select :users::json('{
    "type": "array",
    "items": {
      "type": "object",
      "properties": {
        "name": { "type": "string" },
        "email": { "type": "string", "format": "email" }
      },
      "required": ["name", "email"]
    }
  }') as users
)
select u.name, u.email
from input, json_table(input.users, '$[*]' columns (
  name text path '$.name',
  email varchar(255) path '$.email'
)) as u
```

```sql title="insert_posts.sql"
insert into posts (user_id, title)
with input as (
  select :users::json('{
    "type": "array",
    "items": {
      "type": "object",
      "properties": {
        "email": { "type": "string" },
        "posts": {
          "type": "array",
          "items": {
            "type": "object",
            "properties": { "title": { "type": "string" } },
            "required": ["title"]
          }
        }
      },
      "required": ["email", "posts"]
    }
  }') as users
)
select users.id, u.title
from input, json_table(input.users, '$[*]' columns (
  -- the collation of users.email
  email varchar(255) collate utf8mb4_0900_ai_ci path '$.email',
  nested path '$.posts[*]' columns (title text path '$.title')
)) as u
join users on users.email = u.email
where u.title is not null
```

```ts twoslash
import mysql from "mysql2/promise";
import insertUsers, { type Params } from "./insert_users.sql";
//                         ^?
import insertPosts from "./insert_posts.sql";

const db = mysql.createPool(process.env.DATABASE_URL!);
const users = [
  {
    name: "Ada",
    email: "ada@example.com",
    posts: [{ title: "Notes" }],
  },
];

const conn = await db.getConnection();
try {
  await conn.beginTransaction();
  await insertUsers.run(conn, { users });
  await insertPosts.run(conn, { users });
  await conn.commit();
} catch (error) {
  await conn.rollback();
  throw error;
} finally {
  conn.release();
}
```

A `JSON_TABLE` column takes the connection's collation, `utf8mb4_unicode_ci` under `mysql2`. Comparing it with a column of another collation is an error, so the `email` column names the collation of `users.email`.

</Dialect>

### Run statements in a transaction

A transaction, a savepoint or a connection taken from a client is a client too:

```sql file="rename_user.sql" dialect="all"
update users set name = :name::text where id = :id::int
```

<DialectTabs>
<Dialect value="Postgres">

```ts twoslash
import postgres from "postgres";
import getUser from "./get_user.sql";
import renameUser from "./rename_user.sql";

const db = postgres(process.env.DATABASE_URL!);

await db.begin(async (tx) => {
  const user = await getUser.run(tx, { id: 1 });
  await tx.savepoint((sp) =>
    renameUser.run(sp, { id: user!.id, name: "Grace" }),
  );
});
```

</Dialect>
<Dialect value="MySQL">

```ts twoslash
import mysql from "mysql2/promise";
import renameUser from "./rename_user.sql";

const db = mysql.createPool(process.env.DATABASE_URL!);

const conn = await db.getConnection();
try {
  await conn.beginTransaction();
  await renameUser.run(conn, { id: 1, name: "Grace" });
  await conn.commit();
} catch (error) {
  await conn.rollback();
  throw error;
} finally {
  conn.release();
}
```

</Dialect>
<Dialect value="SQLite">

With `better-sqlite3` or `bun:sqlite`, [`.runSync()`](#runsync) works inside `db.transaction()`:

```ts twoslash
import Database from "better-sqlite3";
import getUser from "./get_user.sql";
import renameUser from "./rename_user.sql";

const db = new Database("app.db");

db.transaction(() => {
  const user = getUser.runSync(db, { id: 1 });
  renameUser.runSync(db, { id: user!.id, name: "Ada Lovelace" });
})();
```

</Dialect>
</DialectTabs>

### Insert or update

When the email already exists, the insert updates that row instead. With `returning`, it resolves to the row's id.

```sql file="upsert_user.sql" tab="Postgres" tab-group="dialect"
insert into users (name, email) values (:name::text, :email::text)
on conflict (email) do update set name = excluded.name
returning id
```

```sql file="upsert_user.sql" tab="MySQL" tab-group="dialect"
insert into users (name, email) values (:name::text, :email::text) as new
on duplicate key update name = new.name
```

```sql file="upsert_user.sql" tab="SQLite" tab-group="dialect"
insert into users (name, email) values (:name::text, :email::text)
on conflict (email) do update set name = excluded.name
returning id
```

```ts twoslash
import upsertUser from "./upsert_user.sql";
```

### Record events of several kinds

A `oneOf` types a JSON parameter as a union. Here a `const` key tells the members apart:

```sql file="record_events.sql" tab="Postgres" tab-group="dialect"
with input as (
  select :events::jsonb('{
    "type": "array",
    "items": {
      "oneOf": [
        {
          "type": "object",
          "properties": {
            "kind": { "type": "string", "const": "click" },
            "x": { "type": "integer" },
            "y": { "type": "integer" }
          },
          "required": ["kind", "x", "y"]
        },
        {
          "type": "object",
          "properties": {
            "kind": { "type": "string", "const": "key" },
            "code": { "type": "string" }
          },
          "required": ["kind", "code"]
        }
      ]
    }
  }') as events
)
insert into events (kind, payload)
select e.value ->> 'kind', e.value
from input, jsonb_array_elements(input.events) as e(value)
```

